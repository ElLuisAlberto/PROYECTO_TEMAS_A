cmake_minimum_required(VERSION 3.8)
project(kinova_direct_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
get_filename_component(KINOVA_DEFAULT "$ENV{HOME}/Escritorio/TESIS_2_DEPENDENCIES/kinova-ros/kinova_driver" ABSOLUTE)
if(DEFINED ENV{KINOVA_ROOT} AND NOT "$ENV{KINOVA_ROOT}" STREQUAL "")
  set(KINOVA_DEFAULT "$ENV{KINOVA_ROOT}")
endif()
set(KINOVA_ROOT "${KINOVA_DEFAULT}" CACHE PATH "Kinova driver directory")
set(KINOVA_LIB_DIR "${KINOVA_ROOT}/lib/x86_64-linux-gnu")
if(NOT EXISTS "${KINOVA_ROOT}/include/kinova/Kinova.API.USBCommandLayerUbuntu.h" OR NOT EXISTS "${KINOVA_LIB_DIR}/USBCommandLayerUbuntu.so")
  message(FATAL_ERROR "Kinova SDK not found; set KINOVA_ROOT")
endif()
add_executable(kinova_direct_node src/kinova_direct_node.cpp)
target_include_directories(kinova_direct_node PRIVATE "${KINOVA_ROOT}/include/kinova")
target_link_directories(kinova_direct_node PRIVATE "${KINOVA_LIB_DIR}")
target_link_libraries(kinova_direct_node "-Wl,-l:USBCommandLayerUbuntu.so")
ament_target_dependencies(kinova_direct_node rclcpp sensor_msgs geometry_msgs std_msgs std_srvs)
set_target_properties(kinova_direct_node PROPERTIES BUILD_RPATH "${KINOVA_LIB_DIR}" INSTALL_RPATH "${KINOVA_LIB_DIR}")
install(TARGETS kinova_direct_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/kinova_cartesian_adapter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scripts/direct_kinematics DESTINATION lib/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
